#define N_MELS            NUM_MEL_FILTERS     // 40, from mel_filterbank.h
#define CONTEXT_FRAMES    NUM_CONTEXT_HOPS    // 100, from feature_matrix.h

// -----------------------------------------------------------------------
// FIX 1: The DSP frontend operates PER HOP (every 10ms), not per 50ms
// batch. The old firmware's `dma_audio_callback` fired every 800 samples
// (50ms) and called a mocked `extract_5_pcen_frames()` that produced 5
// frames at once. The real interface is kws_frontend_process_hop(), called
// once per 160-sample (10ms) hop, with a 400-sample (25ms) OVERLAPPING
// window as input -- 240 of those 400 samples are reused from the previous
// hop. That overlap has to be maintained by US (the embedded/RTOS layer);
// framing.h explicitly says so ("the ring-buffer/streaming side... is
// Embedded C++/RTOS territory").
// -----------------------------------------------------------------------
#define RAW_RING_SAMPLES  (FRAME_SAMPLES * 4)  // headroom for the 400-sample window + slack

static int16_t raw_ring[RAW_RING_SAMPLES];
static volatile int raw_ring_write_pos = 0;
static volatile int raw_samples_available = 0; // new samples since the last hop was extracted

// -----------------------------------------------------------------------
// FIX 2: The DSP pipeline's feature vectors are Q15.16 FIXED-POINT
// (int32_t), NOT int8. The old firmware declared
// `int8_t sliding_pcen_matrix[MATRIX_SIZE]` and memcpy'd straight into the
// TFLM input tensor -- that only works if the DSP output IS int8, which it
// is not. feature_matrix_get_ordered() hands us
// q15_16_t[CONTEXT_FRAMES][N_MELS]; we must QUANTIZE it to int8 ourselves,
// using the model's OWN input scale/zero-point (read from the TFLM input
// tensor at init time), exactly the way the bouncer already dequantizes
// the OUTPUT tensor below. Skipping this step is a silent correctness bug,
// not a crash -- the model would run and produce numbers, just wrong ones.
// -----------------------------------------------------------------------
static q15_16_t ordered_matrix[CONTEXT_FRAMES][N_MELS]; // scratch for feature_matrix_get_ordered()

static float g_input_scale = 1.0f;
static int32_t g_input_zero_point = 0;

static inline int8_t quantize_input_value(q15_16_t raw_q) {
    float f = q15_16_to_float(raw_q);
    int32_t q = (int32_t)lroundf(f / g_input_scale) + g_input_zero_point;
    if (q > 127) q = 127;
    if (q < -128) q = -128;
    return (int8_t)q;
}

// Tensor Arena: internal SRAM, aligned for SIMD. Size here is a PLACEHOLDER --
// re-measure with tflite::RecordingMicroAllocator once the real model + real
// firmware are linked together (this model is bigger than the old draft's
// assumed one: it now genuinely has SubSpectral Norm + freq-pooling, so the
// old 65KB guess needs re-checking, not blindly reused).
static uint8_t tensor_arena[80 * 1024] __attribute__((aligned(16)));

static tflite::MicroInterpreter* interpreter = nullptr;
static TfLiteTensor* input_tensor = nullptr;
static TfLiteTensor* output_tensor = nullptr;

static SemaphoreHandle_t hop_semaphore;
static QueueHandle_t network_event_queue;

typedef struct {
    bool trigger;
    float confidence;
    uint32_t timestamp_ms; // feeds the project's T1 (edge-confirm) latency checkpoint
} KWS_Event;

// -----------------------------------------------------------------------
// I2S/DMA ISR: fires on each DMA buffer completion. Buffer size here should
// be a small, clean divisor of HOP_SAMPLES (160) -- e.g. 160 samples per
// DMA buffer (one hop's worth) keeps this ISR trivially simple. Adjust to
// whatever the actual I2S driver's DMA buffer size ends up being; the ring
// buffer + hop-extraction logic below doesn't care as long as it's fed
// contiguous samples.
// -----------------------------------------------------------------------
void IRAM_ATTR i2s_dma_callback(const int16_t* dma_buffer, size_t n_samples) {
    for (size_t i = 0; i < n_samples; i++) {
        raw_ring[raw_ring_write_pos] = dma_buffer[i];
        raw_ring_write_pos = (raw_ring_write_pos + 1) % RAW_RING_SAMPLES;
    }
    raw_samples_available += (int)n_samples;

    BaseType_t woken = pdFALSE;
    // Only wake the processing task once a full new hop (160 samples) is ready --
    // avoids waking it on every tiny DMA buffer if the DMA buffer is smaller than a hop.
    if (raw_samples_available >= HOP_SAMPLES) {
        xSemaphoreGiveFromISR(hop_semaphore, &woken);
    }
    if (woken) portYIELD_FROM_ISR();
}

// Extracts the most recent FRAME_SAMPLES (400) samples from the ring buffer,
// converts int16 PCM -> Q15.16 (normalized [-1,1) representation), into `out`.
// int16 -> Q15.16: sample/32768 (normalize) * 65536 (Q15.16 scale) == sample * 2,
// i.e. a single left-shift -- cheap, exact, no float involved.
static void extract_q15_frame(q15_16_t* out) {
    int start = (raw_ring_write_pos - FRAME_SAMPLES + RAW_RING_SAMPLES) % RAW_RING_SAMPLES;
    for (int i = 0; i < FRAME_SAMPLES; i++) {
        int idx = (start + i) % RAW_RING_SAMPLES;
        out[i] = ((q15_16_t)raw_ring[idx]) << 1;
    }
}

// -----------------------------------------------------------------------
// FIX 3: All of the old firmware's manual matrix-shifting
// (`memmove(sliding_pcen_matrix, ...)`), the hand-rolled tier0/tier1 gate
// functions, and the "any of 5" hop-counter logic are GONE. kws_frontend.c
// already does all of this internally (ring-buffer feature_matrix with
// O(1) append, tier1 raw-energy gate, tier2 spectral-flatness gate, and the
// confirmed "invoke if ANY of the last 5 hops passed both gates" cadence).
// Reimplementing it here would risk silently drifting from the DSP team's
// tested version. We just call it, once per hop.
// -----------------------------------------------------------------------
static kws_frontend_t frontend;

// ---- Bouncer state (kept from the earlier draft -- this part was already
// correct: a leaky accumulator, not a hard reset, matching our own design
// discussion's recommendation over a strict 3-of-3 reset-on-miss rule) ----
static int consecutive_hits = 0;
static bool is_cooldown = false;
static TickType_t cooldown_start_time = 0;
#define COOLDOWN_MS        1500
#define CONFIDENCE_THRESH  0.85f
#define BOUNCER_TARGET     3

static void run_inference_and_bouncer() {
    if (is_cooldown) {
        if ((xTaskGetTickCount() - cooldown_start_time) < pdMS_TO_TICKS(COOLDOWN_MS)) return;
        is_cooldown = false;
    }

    feature_matrix_get_ordered(&frontend.matrix, ordered_matrix);

    // FIX 2 in action: quantize the Q15.16 matrix into the TFLM input tensor's int8 buffer.
    int8_t* dst = input_tensor->data.int8;
    for (int t = 0; t < CONTEXT_FRAMES; t++) {
        for (int m = 0; m < N_MELS; m++) {
            dst[t * N_MELS + m] = quantize_input_value(ordered_matrix[t][m]);
        }
    }

    interpreter->Invoke();

    float confidence = output_tensor->params.scale *
                        (output_tensor->data.int8[0] - output_tensor->params.zero_point);

    if (confidence > CONFIDENCE_THRESH) {
        consecutive_hits++;
    } else if (consecutive_hits > 0) {
        consecutive_hits--; // leaky decrement, not hard reset -- see design rationale above
    }

    if (consecutive_hits >= BOUNCER_TARGET) {
        consecutive_hits = 0;
        is_cooldown = true;
        cooldown_start_time = xTaskGetTickCount();

        KWS_Event event = { true, confidence, (uint32_t)(xTaskGetTickCount() * portTICK_PERIOD_MS) };
        xQueueSend(network_event_queue, &event, 0);
    }
}

// Main KWS task: one iteration per hop (~10ms), matching kws_frontend's actual granularity.
void kws_main_task(void* pvParameters) {
    kws_frontend_init(&frontend);

    q15_16_t raw_frame[FRAME_SAMPLES];

    while (1) {
        if (xSemaphoreTake(hop_semaphore, portMAX_DELAY)) {
            extract_q15_frame(raw_frame);
            raw_samples_available -= HOP_SAMPLES;

            // One call does tier1 -> (maybe) DSP -> tier2 -> feature_matrix append ->
            // invoke_nn decision. This IS the two-tier gating; there is nothing left
            // for this firmware layer to gate manually.
            kws_hop_result_t result = kws_frontend_process_hop(&frontend, raw_frame);

            if (result.invoke_nn) {
                run_inference_and_bouncer();
            }
        }
    }
}

// Network task: consumes WAKE_CONFIRMED events, wakes Wi-Fi from modem-sleep,
// resumes TLS via cached session ticket, opens the WebSocket audio stream, and
// starts sending the pre-roll + live audio. (Kept high-level here -- the
// lwIP/mbedTLS/WebSocket specifics are the networking subsystem's own file;
// this task's job is just to own the queue and the mode transition, not to
// duplicate that code.)
void network_task(void* pvParameters) {
    KWS_Event event;
    while (1) {
        if (xQueueReceive(network_event_queue, &event, portMAX_DELAY)) {
            // wifi_exit_modem_sleep();

        }
    }
}

// -----------------------------------------------------------------------
// FIX 4: model init now reads the REAL input tensor's scale/zero_point at
// startup (needed for FIX 2's quantization) instead of assuming int8 was
// already produced upstream. Also verifies the model's expected input shape
// matches CONTEXT_FRAMES x N_MELS -- a silent shape mismatch here is exactly
// the kind of bug that only shows up as garbage confidence scores on
// hardware, never as a compile or Invoke() error.
// -----------------------------------------------------------------------
static uint8_t op_resolver_storage[sizeof(tflite::MicroMutableOpResolver<8>)];

bool kws_model_init() {
    static tflite::MicroMutableOpResolver<8>* resolver =
        new (op_resolver_storage) tflite::MicroMutableOpResolver<8>();
    resolver->AddDepthwiseConv2D();
    resolver->AddConv2D();
    resolver->AddAdd();
    resolver->AddMean();      // GlobalAveragePooling2D lowers to Mean
    resolver->AddFullyConnected();
    resolver->AddLogistic();  // sigmoid output
    resolver->AddQuantize();
    resolver->AddDequantize();

    const tflite::Model* model = tflite::GetModel(g_model);
    if (model->version() != TFLITE_SCHEMA_VERSION) return false;

    static tflite::MicroInterpreter static_interpreter(
        model, *resolver, tensor_arena, sizeof(tensor_arena));
    interpreter = &static_interpreter;

    if (interpreter->AllocateTensors() != kTfLiteOk) return false;

    input_tensor = interpreter->input(0);
    output_tensor = interpreter->output(0);

    if (input_tensor->dims->data[1] != CONTEXT_FRAMES || input_tensor->dims->data[2] != N_MELS) {
        return false; // shape mismatch between firmware constants and the trained model
    }

    g_input_scale = input_tensor->params.scale;
    g_input_zero_point = input_tensor->params.zero_point;

    return true;
}

extern "C" void app_main() {
    if (!kws_model_init()) {
        // In production: log the failure reason and halt/retry rather than proceeding
        // with a half-initialized interpreter.
        while (1) { vTaskDelay(pdMS_TO_TICKS(1000)); }
    }

    hop_semaphore = xSemaphoreCreateBinary();
    network_event_queue = xQueueCreate(4, sizeof(KWS_Event));

    // i2s_driver_install(...);           // ESP-IDF I2S/DMA setup (board-specific)
    // i2s_register_dma_callback(i2s_dma_callback);

    xTaskCreatePinnedToCore(kws_main_task, "kws_main", 8192, nullptr, 5, nullptr, 0);
    xTaskCreatePinnedToCore(network_task, "network", 4096, nullptr, 3, nullptr, 1);
}
